Extrai todas as linhas e colunas respeitando aspas e quebras de linha const { rows } = parseFullCsv(csvText); if (rows.length === 0) { rawTeamsData = []; return; } // 2. Identifica se a primeira linha é um cabeçalho const firstRow = rows[0]; const headerKeywords = ['time', 'equipe', 'nome', 'clube', 'team', 'logo', 'link', 'regiao', 'uf', 'estado', 'tag', 'titulo', 'titulos', 'vitoria', 'ponto']; const hasHeader = firstRow.some(col => { const norm = col.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, ""); return headerKeywords.some(k => norm.includes(k)); }); let nameIdx = 0, logoIdx = 1, regionIdx = 2, tagIdx = 3, titlesIdx = 4; let dataRows = rows; if (hasHeader) { dataRows = rows.slice(1); // Remove a linha de cabeçalho apenas se ela existir const headers = firstRow.map(h => h.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "")); const findIdx = (keywords) => headers.findIndex(h => keywords.some(k => h.includes(k))); const nIdx = findIdx(['time', 'equipe', 'nome', 'clube', 'team']); const lIdx = findIdx(['logo', 'link', 'img', 'imagem']); const rIdx = findIdx(['regiao', 'uf', 'estado']); const tgIdx = findIdx(['tag']); const tIdx = findIdx(['titulo', 'titulos', 'vitoria', 'ponto', 'pts']); if (nIdx !== -1) nameIdx = nIdx; if (lIdx !== -1) logoIdx = lIdx; if (rIdx !== -1) regionIdx = rIdx; if (tgIdx !== -1) tagIdx = tgIdx; if (tIdx !== -1) titlesIdx = tIdx; } // 3. Processa cada linha de dados sem descartar a primeira se não houver cabeçalho rawTeamsData = dataRows.map(cols => { const cleanStr = (idx) => (cols[idx] !== undefined ? cols[idx].trim() : ''); let name = cleanStr(nameIdx); // Fallback: se o nome estiver vazio na coluna esperada, procura a primeira coluna com texto válido if (!name) { name = cols.find(c => c && !c.startsWith('http') && isNaN(c) && c.trim().length > 0) || ''; } const logo = cleanStr(logoIdx); const region = cleanStr(regionIdx).toUpperCase(); const tag = cleanStr(tagIdx).toUpperCase(); const titlesRaw = cleanStr(titlesIdx); const titles = parseInt(titlesRaw, 10) || 0; return { name, logo, region, tag, titles }; }).filter(team => team && team.name.trim().length > 0); } // Leitor completo de matriz CSV inteligente function parseFullCsv(csvText) { const rows = []; let curRow = []; let curVal = ''; let inQuotes = false; const text = csvText.replace(/\r\n/g, '\n').replace(/\r/g, '\n'); const firstLine = text.split('\n')[0] || ''; const delimiter = (firstLine.split(';').length > firstLine.split(',').length) ? ';' : ','; for (let i = 0; i < text.length; i++) { const c = text[i]; const next = text[i+1]; if (c === '"') { if (inQuotes && next === '"') { curVal += '"'; i++; } else { inQuotes = !inQuotes; } } else if (c === delimiter && !inQuotes) { curRow.push(curVal.trim()); curVal = ''; } else if (c === '\n' && !inQuotes) { curRow.push(curVal.trim()); if (curRow.some(val => val.length > 0)) { rows.push(curRow); } curRow = []; curVal = ''; } else { curVal += c; } } if (curVal.length > 0 || curRow.length > 0) { curRow.push(curVal.trim()); if (curRow.some(val => val.length > 0)) { rows.push(curRow); } } return { rows, delimiter }; } // --- ATUALIZA A INTERFACE COMPLETA --- function updateUI() { document.getElementById('loading-state').classList.add('hidden'); // Ordenar decrescente por títulos rawTeamsData.sort((a, b) => b.titles - a.titles); renderStats(); renderTabs(); renderTable(); // Atualiza timestamp de checagem const now = new Date(); document.getElementById('last-updated-text').textContent = `Última checagem: ${now.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit', second: '2-digit' })}`; } // --- EXIBIR ESTATÍSTICAS --- function renderStats() { const totalTeams = rawTeamsData.length; const totalTitles = rawTeamsData.reduce((sum, t) => sum + t.titles, 0); const regionsSet = new Set(rawTeamsData.map(t => t.region).filter(Boolean)); const totalRegions = regionsSet.size; const topTeam = rawTeamsData.length > 0 ? rawTeamsData[0].name : '-'; document.getElementById('stat-total-teams').textContent = totalTeams; document.getElementById('stat-total-titles').textContent = totalTitles; document.getElementById('stat-total-regions').textContent = totalRegions; document.getElementById('stat-top-team').textContent = topTeam; } // --- GERAR ABAS DINÂMICAS (GERAL + REGIÕES) --- function renderTabs() { const tabsNav = document.getElementById('tabs-nav'); tabsNav.innerHTML = ''; // 1. Obter lista de regiões únicas presentes nos dados const regions = Array.from(new Set(rawTeamsData.map(t => t.region).filter(Boolean))).sort(); // Se a aba ativa atual não existe mais nas regiões e não é GERAL, reseta para GERAL if (currentActiveTab !== 'GERAL' && !regions.includes(currentActiveTab)) { currentActiveTab = 'GERAL'; } // Lista de abas a serem criadas const tabList = [ { id: 'GERAL', label: 'Geral', count: rawTeamsData.length }, ...regions.map(r => ({ id: r, label: `Região ${r}`, count: rawTeamsData.filter(t => t.region === r).length })) ]; tabList.forEach(tab => { const isActive = currentActiveTab === tab.id; const btn = document.createElement('button'); btn.className = `px-4 py-2 text-xs font-semibold rounded-xl transition duration-150 flex items-center space-x-2 shrink-0 ${ isActive ? 'bg-blue-600 text-white shadow-md shadow-blue-600/30' : 'bg-dark-card text-gray-400 hover:text-white border border-dark-border hover:bg-dark-hover' }`; btn.innerHTML = ` ${tab.label} ${tab.count} `; btn.addEventListener('click', () => { currentActiveTab = tab.id; renderTabs(); renderTable(); }); tabsNav.appendChild(btn); }); } // --- RENDERIZAR TABELA DE RANKING --- function renderTable() { const tbody = document.getElementById('ranking-tbody'); const emptyState = document.getElementById('empty-state'); tbody.innerHTML = ''; // 1.
